## How to use

```tsx
import { NavLine } from "@/components/hangry-ui/components/nav-line";

<NavLine
  items={[
    { id: "overview", label: "Overview" },
    { id: "usage", label: "Usage", card: {
      title: "Usage",
      description: "Examples and configuration.",
    } },
  ]}
/>

<section id="overview">Overview content</section>
<section id="usage">Usage content</section>
```
