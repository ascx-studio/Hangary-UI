## How to use

```tsx
import { CardStack } from "@/components/hangry-ui/components/card-stack";

<CardStack />

<CardStack items={[
  { id: "first", src: "/1.png", alt: "Glowing butterflies", title: "Wings of light", description: "A different perspective." },
  { id: "second", src: "/2.png", alt: "Swirling white light", title: "Into the orbit" },
]} />
```
