## How to use

```tsx
"use client";

import type { ComponentProps } from "react";
import { LoginSplit } from "@/components/hangry-ui/blocks/login-split";

// Pass your application's authentication handlers to this form.
export function SignIn(
  props: Pick<
    ComponentProps<typeof LoginSplit>,
    "onSubmit" | "onProviderLogin" | "loading" | "error"
  >,
) {
  return <LoginSplit brand="Acme" {...props} />;
}
```
