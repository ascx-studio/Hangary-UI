## How to use

```tsx
import { FooterStacked } from "@/components/hangry-ui/blocks/footer-stacked";

<FooterStacked
  brand="Acme"
  description="Tools for your next idea."
  copyright="© 2026 Acme"
  githubHref="#github"
  groups={[
    {
      title: "Pages",
      links: [
        { label: "Home", href: "#home" },
        { label: "Components", href: "#components" },
        { label: "Blocks", href: "#blocks" },
      ],
    },
    {
      title: "More",
      links: [
        { label: "Shaders", href: "#shaders" },
        { label: "Sponsor", href: "#sponsor" },
        { label: "Privacy", href: "#privacy" },
      ],
    },
  ]}
  callout={{
    title: "Build something new",
    description: "Explore the component collection.",
    label: "Get started",
    href: "#components",
  }}
/>;
```
