# Login

A card login form with provider buttons, password visibility, and loading and error states.

## Installation

<CodeBlock commands={{
  bun: "bunx shadcn@latest add https://ui.razeev.com/r/login.json",
  npm: "npx shadcn@latest add https://ui.razeev.com/r/login.json",
  pnpm: "pnpm dlx shadcn@latest add https://ui.razeev.com/r/login.json",
  yarn: "yarn dlx shadcn@latest add https://ui.razeev.com/r/login.json",
}} />

## Usage

```tsx
"use client";

import type { ComponentProps } from "react";
import { Login } from "@/components/hangry-ui/blocks/login";

// Pass your application's authentication handlers to this form.
export function SignIn(props: Pick<ComponentProps<typeof Login>,
  "onSubmit" | "onProviderLogin" | "loading" | "error">) {
  return <Login brand="Acme" {...props} />;
}
```
