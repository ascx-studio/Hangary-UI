## How to use

```tsx
"use client";

import type { ComponentProps } from "react";
import { Login } from "@/components/hangry-ui/blocks/login";

// Pass your application's authentication handlers to this form.
export function SignIn(
  props: Pick<ComponentProps<typeof Login>, "onSubmit" | "onProviderLogin" | "loading" | "error">,
) {
  return <Login brand="Acme" {...props} />;
}
```
