## How to use

```tsx
import { FaqCards } from "@/components/hangry-ui/blocks/faq-cards";

<FaqCards />

<FaqCards
  title="Good questions. Clear answers."
  description="Everything you need to know before getting started."
  defaultOpenId="customize"
  items={[
    { id: "customize", question: "Can I use my own content?", answer: "Yes. Replace the questions and answers with your own." },
    { id: "support", question: "Where can I find help?", answer: <a href="/contact" className="underline">Contact our team</a> },
  ]}
/>
```
